Consultas utilizadas nos indicadores

Eficiência mensal por setor

CTE + agregação
WITH producao_setor AS (
  SELECT s.nome_setor,
         SUM(p.produzido) produzido,
         SUM(p.meta) meta
  FROM fato_producao p
  JOIN dim_setores s
    ON s.id_setor = p.id_setor
  WHERE p.data_producao
    BETWEEN '2026-09-01'
    AND '2026-09-30'
  GROUP BY s.nome_setor
)
SELECT nome_setor,
  produzido * 100.0 / NULLIF(meta, 0)
  AS eficiencia_pct
FROM producao_setor
ORDER BY eficiencia_pct DESC;

Pareto das paradas

window function
SELECT motivo,
       SUM(duracao_minutos) minutos,
       SUM(SUM(duracao_minutos))
         OVER (ORDER BY SUM(duracao_minutos) DESC)
         * 100.0 /
       SUM(SUM(duracao_minutos))
         OVER () AS percentual_acumulado
FROM fato_paradas
WHERE data BETWEEN '2026-09-01'
                   AND '2026-09-30'
GROUP BY motivo
ORDER BY minutos DESC;
